"use client";
import { useState } from "react";
import BuscaCep from "./BuscaCep";

export default function PaginaCep() {
  const [cep, setCep] = useState("");
  const [aviso, setAviso] = useState("");
  const [endereco, setEndereco] = useState(null);

  async function enviarCep(e) {
    e.preventDefault();

    const cepLimpo = cep.replace(/\D/g, "");

    if (cepLimpo.length !== 8) {
      setAviso("Digite um CEP com 8 números.");
      return;
    }

    try {
      const res = await fetch(`https://viacep.com.br/ws/${cepLimpo}/json/`);
      const dados = await res.json();

      if (dados.erro) {
        setAviso("CEP não encontrado.");
        setEndereco(null);
        return;
      }

      setEndereco(dados);
      setAviso("");
    } catch {
      setAviso("Erro ao buscar o CEP. Tente de novo.");
    }
  }

  return (
    <div>
      <form onSubmit={enviarCep}>
        <input
          value={cep}
          onChange={(e) => setCep(e.target.value)}
          placeholder="Digite seu CEP"
        />
        <button type="submit">Enviar</button>
      </form>

      <p>{aviso}</p>

      {endereco && <BuscaCep endereco={endereco} />}
    </div>
  );
}


===== .\app\componentes\Tarefas.js =====

"use client";
import { useState, useEffect } from "react";
import {
  listarTarefas,
  criarTarefa,
  marcarTarefa,
  removerTarefa,
} from "../lib/tarefas.js";

export default function Tarefas() {
  const [tarefas, setTarefas] = useState([]);
  const [texto, setTexto] = useState("");

  useEffect(() => {
    listarTarefas().then(setTarefas);
  }, []);

  async function criar(e) {
    e.preventDefault();
    if (texto.trim() === "") return;

    const nova = await criarTarefa(texto);
    setTarefas([...tarefas, nova]);
    setTexto("");
  }

  async function marcar(id) {
    const tarefa = tarefas.find((t) => t.id === id);
    await marcarTarefa(id, !tarefa.feita);
    setTarefas(
      tarefas.map((t) => (t.id === id ? { ...t, feita: !t.feita } : t))
    );
  }

  async function remover(id) {
    await removerTarefa(id);
    setTarefas(tarefas.filter((t) => t.id !== id));
  }


  return (
    <div>
      <form onSubmit={criar}>
        <input value={texto} onChange={(e) => setTexto(e.target.value)} />
        <button type="submit">Adicionar</button>
      </form>

      <ul>
        {tarefas.map((t) => (
          <li key={t.id}>
            <input
              type="checkbox"
              checked={t.feita}
              onChange={() => marcar(t.id)}
            />
            {t.texto}
            <button onClick={() => remover(t.id)}>Remover</button>
          </li>
        ))}
      </ul>
    </div>
  );
}

"use client";
import { useState } from "react";
import { salvarMensagem } from "../lib/Contato";

export default function Contato() {
  const [nome, setNome] = useState("");
  const [mensagem, setMensagem] = useState("");
  const [aviso, setAviso] = useState("");

  async function enviar(e) {
    e.preventDefault();

    if (nome.trim().length < 3) {
      setAviso("Digite seu nome completo.");
      return;
    }
    if (mensagem.trim().length < 5) {
      setAviso("Escreva uma mensagem um pouco maior.");
      return;
    }

    try {
      await salvarMensagem(nome.trim(), mensagem.trim());
      setAviso(`Obrigado, ${nome}! Recebi sua mensagem.`);
      setNome("");
      setMensagem("");
    } catch (erro) {
      setAviso("Não foi possível enviar. Tente de novo.");
    }
  }

  return (
    <><div>
      <form onSubmit={enviar}>
        <div>
          <input
            value={nome}
            onChange={(e) => setNome(e.target.value)}
            placeholder="Seu nome" />
        </div>
        <div>
          <textarea
            value={mensagem}
            onChange={(e) => setMensagem(e.target.value)}
            placeholder="Sua mensagem"
            rows={4} />
        </div>
        <button type="submit">Enviar</button>
      </form>
      <p>{aviso}</p>
    </div>
    <div>
        <p>Contato:</p>
        <p>gabriel@bravilsistemas.com</p>
    </div></>
  );
}

"use client";
import { useState} from "react";

export default function ListaTarefas() {
  const [tarefas, setTarefas] = useState([
    { id: 1, texto: "Estudar useState", feita: false },
    { id: 2, texto: "Fazer o formulário de contato", feita: true },
  ]);
  const [novaTarefa, setNovaTarefa] = useState("");

  function adicionar(e) {
    e.preventDefault();
    if (novaTarefa.trim() === "") return;
    setTarefas([...tarefas, { id: Date.now(), texto: novaTarefa, feita: false }]);
    setNovaTarefa("");
  }

  function alternar(id) {
    setTarefas(
      tarefas.map((t) => (t.id === id ? { ...t, feita: !t.feita } : t))
    );
  }

  const faltam = tarefas.filter((t) => !t.feita).length;
  const total = tarefas.length;
  const percentual =
    total === 0 ? 0 : Math.round(((total - faltam) / total) * 100);

  return (
    <section className="max-w-md mx-auto my-10 p-6 rounded-2xl bg-superficie shadow-lg border border-destaque/20">
      <div className="flex items-center justify-between mb-4">
        <h2 className="text-xl font-bold text-destaque">Minhas tarefas</h2>
        <span className="text-sm font-medium px-3 py-1 rounded-full bg-destaque/10 text-destaque">
          {faltam === 0 ? "Tudo feito!" : `Faltam ${faltam} de ${total}`}
        </span>
      </div>

      <div className="h-2 mb-6 rounded-full bg-destaque/10 overflow-hidden">
        <div
          className="h-full rounded-full bg-destaque transition-all duration-500"
          style={{ width: `${percentual}%` }}
        />
      </div>

      <form onSubmit={adicionar} className="flex gap-2 mb-6">
        <input
          value={novaTarefa}
          onChange={(e) => setNovaTarefa(e.target.value)}
          placeholder="Nova tarefa"
          className="flex-1 px-4 py-2 rounded-lg border border-destaque/30 bg-white outline-none focus:ring-2 focus:ring-destaque/40 placeholder:text-gray-400"
        />
        <button
          type="submit"
          className="px-4 py-2 rounded-lg bg-destaque text-white font-medium hover:opacity-90 active:scale-95 transition"
        >
          Adicionar
        </button>
      </form>

      <ul className="space-y-2">
        {tarefas.map((t) => (
          <li
            key={t.id}
            className="rounded-lg hover:bg-destaque/5 transition-colors"
          >
            <label className="flex items-center gap-3 px-3 py-2 cursor-pointer">
              <input
                type="checkbox"
                checked={t.feita}
                onChange={() => alternar(t.id)}
                className="size-5 accent-destaque cursor-pointer"
              />
              <span
                className={`transition-colors ${
                  t.feita ? "line-through text-gray-400" : ""
                }`}
              >
                {t.texto}
              </span>
            </label>
          </li>
        ))}
      </ul>

      {total === 0 && (
        <p className="text-center text-gray-400 py-4">
          Nenhuma tarefa por enquanto.
        </p>
      )}
    </section>
  );
}

===== .\lib\tarefas.js =====

import { db } from "./firebase";
import {
  collection,
  addDoc,
  getDocs,
  updateDoc,
  deleteDoc,
  doc,
  query,
  orderBy,
} from "firebase/firestore";

const colecao = collection(db, "tarefas");

export async function listarTarefas() {
  const consulta = query(colecao, orderBy("criadaEm"));
  const resultado = await getDocs(consulta);
  return resultado.docs.map((d) => ({ id: d.id, ...d.data() }));
}

export async function criarTarefa(texto) {
  const dados = { texto: texto, feita: false, criadaEm: Date.now() };
  const ref = await addDoc(colecao, dados);
  return { id: ref.id, ...dados };
}

export async function marcarTarefa(id, feita) {
  await updateDoc(doc(db, "tarefas", id), { feita: feita });
}

export async function removerTarefa(id) {
  await deleteDoc(doc(db, "tarefas", id));
}
